def make_panel(tasks, horizon=2400):
    """按时段、区域和任务类型汇总任务数量及 GPU 需求。"""
    panel = (
        tasks.groupby(["hour", "region", "task_type"])
        .agg(
            task_count=("task_id", "count"),
            gpu_demand=("gpu_num", "sum")
        )
        .reset_index()
    )

    full_index = pd.MultiIndex.from_product(
        [range(horizon), panel["region"].unique(), panel["task_type"].unique()],
        names=["hour", "region", "task_type"]
    )

    panel = (
        panel.set_index(["hour", "region", "task_type"])
        .reindex(full_index, fill_value=0)
        .reset_index()
    )
    return panel

def candidate_prediction(train_series, horizon, mode):
    """生成不同时间序列预测方法对应的预测值。"""
    if mode == "direct_mean":
        value = train_series.mean()
        return np.repeat(value, horizon)

    if mode == "rolling_mean":
        value = train_series.tail(168).mean()
        return np.repeat(value, horizon)

    if mode == "daily_seasonal":
        pattern = train_series.tail(24).to_numpy()
        return np.resize(pattern, horizon)

    if mode == "weekly_seasonal":
        pattern = train_series.tail(168).to_numpy()
        return np.resize(pattern, horizon)

    raise ValueError("未知预测模式")

def score_predictions(actual, prediction):
    """计算 MAE、RMSE、WAPE 和系统性偏差。"""
    error = actual - prediction
    mae = np.mean(np.abs(error))
    rmse = np.sqrt(np.mean(error ** 2))
    wape = np.sum(np.abs(error)) / max(np.sum(np.abs(actual)), 1e-8)
    bias = np.mean(error)

    return {
        "MAE": mae,
        "RMSE": rmse,
        "WAPE": wape,
        "Bias": bias
    }

def solve_lexicographic(model):
    """先最小化峰值算力占用，再最小化加权等待时间。"""

    # 第一阶段：削减峰值 GPU 使用率
    peak_result = solve_stage(
        model=model,
        objective="peak_gpu_utilization"
    )

    peak_cap = peak_result["peak_gpu_utilization"] * 1.001

    # 第二阶段：在峰值近似最优条件下改善服务水平
    wait_result = solve_stage(
        model=model,
        objective="weighted_waiting_time",
        extra_constraints=[
            ("peak_gpu_utilization", "<=", peak_cap)
        ]
    )

    return wait_result

5. 基础约束核验
def check_schedule(schedule, gpu_capacity, task_info):
    """核验任务唯一分配、到达时刻及区域算力容量约束。"""
    violations = []

    for task_id, group in schedule.groupby("task_id"):
        if len(group) != 1:
            violations.append(f"任务{task_id}未唯一分配")

    for _, row in schedule.iterrows():
        if row["start_hour"] < task_info[row["task_id"]]["arrival"]:
            violations.append(f"任务{row['task_id']}早于到达时刻执行")

    for region, load in schedule.groupby(["region", "hour"])["gpu_num"].sum().items():
        if load > gpu_capacity[region]:
            violations.append(f"{region}发生算力容量超限")

    return violations

1. 电力平衡与碳排放核算
def energy_dispatch(load, renewable, export_cap):
    """无储能条件下的逐时电力平衡计算。"""
    direct_renew = np.minimum(load, renewable)
    grid_purchase = np.maximum(load - renewable, 0)

    renewable_surplus = np.maximum(renewable - load, 0)
    grid_sale = np.minimum(renewable_surplus, export_cap)
    curtailment = renewable_surplus - grid_sale

    return {
        "direct_renew": direct_renew,
        "grid_purchase": grid_purchase,
        "grid_sale": grid_sale,
        "curtailment": curtailment
    }

def calculate_metrics(dispatch, price, carbon_factor):
    """计算购电成本、售电收益及实际碳排放。"""
    purchase_cost = np.sum(dispatch["grid_purchase"] * price)
    sale_revenue = np.sum(dispatch["grid_sale"] * price)
    carbon_emission = np.sum(
        dispatch["grid_purchase"] * carbon_factor
    )

    return {
        "net_cost": purchase_cost - sale_revenue,
        "carbon_emission": carbon_emission
    }

def construct_feasible_schedule(tasks, capacity_state):
    """按截止期紧迫度优先构造满足硬约束的初始排程。"""
    sorted_tasks = sorted(
        tasks,
        key=lambda x: (
            x["deadline"] - x["arrival"] - x["duration"],
            -x["gpu_hours"]
        )
    )

    schedule = []
    for task in sorted_tasks:
        feasible_starts = generate_feasible_starts(
            task=task,
            capacity_state=capacity_state
        )

        if len(feasible_starts) == 0:
            raise RuntimeError(f"任务{task['task_id']}无可行开工时刻")

        start = min(feasible_starts)
        assign_task(task, start, capacity_state)
        schedule.append({
            "task_id": task["task_id"],
            "region": task["region"],
            "start_hour": start
        })

    return pd.DataFrame(schedule)

3. 局部窗口任务筛选
def select_window_tasks(schedule, tasks, start_hour, end_hour, max_tasks=120):
    """筛选完全位于局部时间窗口内的可迁移任务。"""
    selected = []

    for task in tasks:
        task_start = schedule.loc[
            schedule["task_id"] == task["task_id"], "start_hour"
        ].iloc[0]

        task_end = task_start + task["duration"]

        if task_start >= start_hour and task_end <= end_hour:
            selected.append(task)

    selected = sorted(
        selected,
        key=lambda x: x["gpu_num"] * x["duration"],
        reverse=True
    )

    return selected[:max_tasks]

4. 局部 MILP 精修模型
def local_mip_polish(local_tasks, candidate_starts, fixed_load, capacity):
    """固定窗口外负荷，对窗口内任务进行重排。"""
    model = gp.Model("local_schedule")

    x = {}
    for task in local_tasks:
        for start in candidate_starts[task["task_id"]]:
            x[task["task_id"], start] = model.addVar(
                vtype=GRB.BINARY,
                name=f"x_{task['task_id']}_{start}"
            )

    # 每项任务唯一开工
    for task in local_tasks:
        model.addConstr(
            gp.quicksum(
                x[task["task_id"], start]
                for start in candidate_starts[task["task_id"]]
            ) == 1
        )

    # 时段算力容量约束
    for hour in range(len(capacity)):
        local_gpu = gp.quicksum(
            task["gpu_num"] * x[task["task_id"], start]
            for task in local_tasks
            for start in candidate_starts[task["task_id"]]
            if start <= hour < start + task["duration"]
        )

        model.addConstr(fixed_load[hour] + local_gpu <= capacity[hour])

    return model, x

def slot_proxy(hour, carbon, renewable, price, peak_pressure):
    """衡量任务置于某时段的相对优劣。"""
    score = (
        0.35 * carbon[hour]
        - 0.30 * renewable[hour]
        + 0.20 * price[hour]
        + 0.15 * peak_pressure[hour]
    )
    return score

2. 候选开工时刻生成
def candidate_starts(task, proxy_score, top_k=3):
    """保留边界时刻、分位时刻和代理评分最优时刻。"""
    earliest = task["arrival"]
    latest = task["deadline"] - task["duration"]

    feasible = list(range(earliest, latest + 1))
    ranked = sorted(feasible, key=lambda t: proxy_score[t])

    candidates = {
        earliest,
        latest,
        feasible[len(feasible) // 3],
        feasible[2 * len(feasible) // 3]
    }
    candidates.update(ranked[:top_k])

    return sorted(candidates)

3. 局部邻域重调度
def choose_neighbourhood(tasks, schedule, top_n=80):
    """优先选择能耗高、可迁移范围大的任务参与局部优化。"""
    scored_tasks = []

    for task in tasks:
        flexibility = task["deadline"] - task["arrival"] - task["duration"]
        impact = task["gpu_num"] * task["duration"]

        scored_tasks.append({
            "task": task,
            "score": impact * max(flexibility, 1)
        })

    scored_tasks = sorted(
        scored_tasks,
        key=lambda x: x["score"],
        reverse=True
    )
    return [x["task"] for x in scored_tasks[:top_n]]

def dominates(solution_a, solution_b):
    """判断方案 a 是否支配方案 b。"""
    indicators = ["net_cost", "weighted_wait", "carbon_emission"]

    no_worse = all(
        solution_a[k] <= solution_b[k]
        for k in indicators
    )

    strictly_better = any(
        solution_a[k] < solution_b[k]
        for k in indicators
    )

    return no_worse and strictly_better
