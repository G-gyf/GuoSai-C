**2. 逐区间电力平衡与 SOC 状态方程**

```python
def _build_equalities(load_kwh, pv_kwh, parameters):
    """构建电力平衡与储能荷电状态转移的等式约束矩阵。"""
    n = len(load_kwh)
    variable_count = 5 * n
    grid, charge, discharge, curtailment, soc = 0, n, 2 * n, 3 * n, 4 * n
    matrix = np.zeros((2 * n + 1, variable_count), dtype=float)
    rhs = np.zeros(2 * n + 1, dtype=float)

    # 逐区间电力平衡：购电 + 光伏 = 负载 + 充电 - 放电 + 弃光
    for t in range(n):
        matrix[t, grid + t] = 1.0
        matrix[t, charge + t] = -1.0
        matrix[t, discharge + t] = 1.0
        matrix[t, curtailment + t] = -1.0
        rhs[t] = load_kwh[t] - pv_kwh[t]

    # SOC 转移：E_t + eta_c*C_t - D_t/eta_d = E_{t+1}；首区间由初始电量确定
    for t in range(n):
        row = n + t
        matrix[row, soc + t] = 1.0
        matrix[row, charge + t] = -parameters.charge_efficiency
        matrix[row, discharge + t] = 1.0 / parameters.discharge_efficiency
        if t == 0:
            rhs[row] = parameters.initial_soc_kwh
        else:
            matrix[row, soc + t - 1] = -1.0

    # 0:00 与 24:00 储电量相等（问题 1 硬性要求）
    matrix[-1, soc + n - 1] = 1.0
    rhs[-1] = parameters.terminal_soc_kwh
    return matrix, rhs
```

**3. 变量边界设置**

```python
def _variable_bounds(n, parameters, pv_kwh):
    """购电量、充放电量、弃光量与 SOC 的非负性及上下界。"""
    return (
        [(0.0, None)] * n                                   # 购电量 g
        + [(0.0, parameters.max_charge_kwh)] * n            # 充电量 C <= 5000kW*1/6h
        + [(0.0, parameters.max_discharge_kwh)] * n         # 放电量 D
        + [(0.0, float(value)) for value in pv_kwh]         # 弃光量 U <= 光伏功率
        + [(parameters.min_soc_kwh, parameters.max_soc_kwh)] * n  # 1200 <= SOC <= 10800
    )
```

**4. 两阶段字典序线性规划**

```python
def _solve_main_two_stage_lp(inputs, parameters):
    """第一阶段最小化购电费；第二阶段在成本近似最优下最小化充放吞吐。"""
    n = len(inputs)
    price = inputs["price_yuan_per_kwh"].to_numpy(dtype=float)
    load_kwh = inputs["load_kw"].to_numpy(dtype=float) * parameters.interval_hours
    pv_kwh = inputs["pv_kw"].to_numpy(dtype=float) * parameters.interval_hours
    equalities, equality_rhs = _build_equalities(load_kwh, pv_kwh, parameters)
    bounds = _variable_bounds(n, parameters, pv_kwh)

    # 第一阶段：最小化全天购电费
    cost_objective = np.zeros(5 * n, dtype=float)
    cost_objective[:n] = price
    first = linprog(
        cost_objective,
        A_eq=equalities,
        b_eq=equality_rhs,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not first.success:
        raise RuntimeError(f"Question 1 main cost LP failed: {first.message}")

    # 第二阶段：成本容差内最小化充放电总吞吐，消除等成本解中的无效同时充放
    cost_tolerance = max(1e-7, 1e-10 * abs(float(first.fun)))
    tie_break_cost_cap = float(first.fun) + cost_tolerance
    throughput_objective = np.zeros(5 * n, dtype=float)
    throughput_objective[n : 3 * n] = 1.0
    second = linprog(
        throughput_objective,
        A_ub=cost_objective.reshape(1, -1),
        b_ub=np.asarray([tie_break_cost_cap]),
        A_eq=equalities,
        b_eq=equality_rhs,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not second.success:
        raise RuntimeError(f"Question 1 main throughput LP failed: {second.message}")
    return second.x
```

```python
def validate_solution(schedule, summary, parameters):
    """逐项核验平衡、SOC、边界、同时充放与费用复算。"""
    tol = parameters.feasibility_tolerance
    balance = (
        schedule["grid_purchase_kwh"] + schedule["pv_kwh"] + schedule["discharge_kwh"]
        - schedule["load_kwh"] - schedule["charge_kwh"] - schedule["curtailment_kwh"]
    )
    soc_transition = (
        schedule["soc_end_kwh"] - schedule["soc_start_kwh"]
        - parameters.charge_efficiency * schedule["charge_kwh"]
        + schedule["discharge_kwh"] / parameters.discharge_efficiency
    )
    simultaneous = (schedule["charge_kwh"] > tol) & (schedule["discharge_kwh"] > tol)
    checks = {
        "max_balance_residual_kwh": float(balance.abs().max()),
        "max_soc_transition_residual_kwh": float(soc_transition.abs().max()),
        "min_soc_kwh": float(schedule["soc_end_kwh"].min()),
        "max_soc_kwh": float(schedule["soc_end_kwh"].max()),
        "terminal_soc_kwh": float(schedule["soc_end_kwh"].iloc[-1]),
        "max_charge_kwh": float(schedule["charge_kwh"].max()),
        "max_discharge_kwh": float(schedule["discharge_kwh"].max()),
        "simultaneous_charge_discharge_slots": int(simultaneous.sum()),
        "cost_recalculation_error_yuan": float(
            abs(
                summary["total_cost_yuan"]
                - np.dot(schedule["price_yuan_per_kwh"], schedule["grid_purchase_kwh"])
            )
        ),
    }
    failures = []
    if checks["max_balance_residual_kwh"] > tol:
        failures.append("power_balance")
    if checks["max_soc_transition_residual_kwh"] > tol:
        failures.append("soc_transition")
    if abs(checks["terminal_soc_kwh"] - parameters.terminal_soc_kwh) > tol:
        failures.append("terminal_soc")
    if checks["simultaneous_charge_discharge_slots"]:
        failures.append("simultaneous_charge_discharge")
    if checks["cost_recalculation_error_yuan"] > tol:
        failures.append("cost_recalculation")
    # 其余边界检查（SOC 上下限、充放电上限、购电/弃光非负等）同此
    if failures:
        raise RuntimeError(f"Question 1 validation failed: {', '.join(failures)}")
    return checks
```

```python
def equality_matrix(n):
    """变量 g, C, D, U, E（kWh）：平衡方程与 SOC 转移方程。"""
    a = lil_matrix((2 * n, 5 * n))
    for t in range(n):
        a[t, t], a[t, n + t], a[t, 2 * n + t], a[t, 3 * n + t] = 1, -1, 1, -1
        a[n + t, 4 * n + t], a[n + t, n + t], a[n + t, 2 * n + t] = 1, -ETA, 1 / ETA
        if t:
            a[n + t, 4 * n + t - 1] = -1
    return a.tocsr()


def solve_plan(net, prices, initial, terminal_value):
    """日前计划 LP：最小化计划购电费 - 期末电量价值，再消除无效吞吐。"""
    n = len(net)
    a = equality_matrix(n)
    b = np.r_[net, initial, np.zeros(n - 1)]
    bounds = [(0, None)] * n + [(0, S)] * (2 * n) + [(0, None)] * n + [(EMIN, EMAX)] * n
    obj = np.zeros(5 * n)
    obj[:n], obj[-1] = prices, -terminal_value      # 谷值替代成本 nu = min(p)/eta
    first = linprog(obj, A_eq=a, b_eq=b, bounds=bounds, method='highs')
    if not first.success:
        raise RuntimeError(first.message)
    throughput = np.zeros(5 * n)
    throughput[n:3 * n] = 1
    second = linprog(
        throughput, A_eq=a, b_eq=b, bounds=bounds,
        A_ub=csr_matrix(obj.reshape(1, -1)), b_ub=[first.fun + 1e-6], method='highs')
    if not second.success:
        raise RuntimeError(second.message)
    x = second.x.copy()
    x[np.abs(x) < 1e-8] = 0
    return x.reshape(5, n), float(first.fun)
```

**4. 逐区间物理转移规则**

```python
def rule_transition(net, grid, energy, reserve):
    """情景评估与真实执行共用的单区间物理转移。"""
    r = np.asarray(net, dtype=float) - np.asarray(grid, dtype=float)
    e = np.asarray(energy, dtype=float)
    target = np.asarray(reserve, dtype=float)
    charge = np.minimum(
        np.minimum(np.maximum(-r, 0.0), S),
        np.maximum((EMAX - e) / ETA, 0.0),
    )
    discharge = np.minimum(
        np.minimum(np.maximum(r, 0.0), S),
        ETA * np.maximum(e - target, 0.0),
    )
    emergency = np.maximum(r - discharge, 0.0)
    unused = np.maximum(-r - charge, 0.0)
    end = e + ETA * charge - discharge / ETA
    return charge, discharge, emergency, unused, end
```

def calibrate_rule(net_paths, forecast_net, grid, plan_soc, prices, initial,
                   terminal_value, residual_count, settings, seed):
    """在最近 21 天历史情景上差分进化搜索 7 参数，劣于零参数则保留零。"""
    objective = ScenarioObjective(
        net_paths, forecast_net, grid, plan_soc, prices, initial, terminal_value)
    zero = np.zeros(7, dtype=float)
    zero_score = objective(zero)
    if residual_count < settings.residual_days:
        return zero
    bounds = ([(-settings.delta_bound_kwh, settings.delta_bound_kwh)] * 4
              + [(-settings.lambda_bound, settings.lambda_bound)] * 3)
    result = differential_evolution(
        objective, bounds, strategy="best1bin",
        maxiter=settings.search_maxiter, popsize=settings.search_popsize,
        seed=seed, polish=False, init="latinhypercube",
        updating="deferred", vectorized=True, x0=zero,
    )
    candidate = np.asarray(result.x, dtype=float)
    candidate_score = float(result.fun)
    accepted = candidate_score <= zero_score + 1e-8
    return candidate if accepted else zero


def execute_actual_day(load, pv, grid, plan_soc, prices, initial, forecast_net, theta):
    """逐日因果执行：每个阶段起点锁定信号，逐区间推进真实 SOC。"""
    delta, lambdas = unpack_theta(theta)
    energy = float(initial)
    error_sum, signal = 0.0, 0.0
    out = {key: np.zeros(T, dtype=float) for key in [
        "reserve_kwh", "stage_error_mean_kwh", "charge_kwh", "discharge_kwh",
        "emergency_kwh", "unused_kwh", "soc_end_kwh", "emergency_cost"]}
    actual_net = load - pv
    for t in range(T):
        stage = min(t // STAGE_LENGTH, 3)
        if t % STAGE_LENGTH == 0:
            signal = 0.0 if t == 0 else error_sum / t
        reserve = float(np.clip(
            plan_soc[t] + delta[stage] + lambdas[stage] * signal, EMIN, EMAX))
        charge, discharge, emergency, unused, energy = rule_transition(
            actual_net[t], grid[t], energy, reserve)
        out["charge_kwh"][t], out["discharge_kwh"][t] = charge, discharge
        out["emergency_kwh"][t], out["unused_kwh"][t] = emergency, unused
        out["soc_end_kwh"][t] = energy
        out["emergency_cost"][t] = 5.0 * prices[t] * emergency
        error_sum += actual_net[t] - forecast_net[t]
    return out

def settle(q0, qA, prices):
    """保留部分、下调违约（0.5c）与上调溢价（1.5c）的逐区间结算分解。"""
    q0 = np.asarray(q0, float)
    qA = np.asarray(qA, float)
    prices = np.asarray(prices, float)
    retained = prices * np.minimum(q0, qA)
    down = 0.5 * prices * np.maximum(q0 - qA, 0.0)
    up = 1.5 * prices * np.maximum(qA - q0, 0.0)
    return retained, down, up

def scenario_matrix(d, fl, fc, load, pv, k, h0, residual_days=21):
    """发布 k 在 h0 时刻对未来 [h0,144) 区间的因果净负荷情景。"""
    first = max(7, d - residual_days)
    count = d - first
    if count <= 0:
        return None, 0
    eps_l = load[first:d] - fl[first:d]
    eps_v = pv[first:d] - fc[first:d, k]

    sv = np.maximum(0.0, fc[d, k, h0:] + eps_v[:, h0:])
    return sl - sv, count

```python
def adjustment_curve(fl_d, fc_dk, scen, q0, h0, q_up=0.7, q_down=0.9):
    """滚动调整的双面报童曲线：median(Q70, Q90, q0)，点预测作下限。

    分段净结算下，下调边际成本 0.5c、上调边际成本 1.5c，由单区间无储能
    报童条件 R'(a) = 5c * P(D > a) 得覆盖水平
        F_down = 1 - 0.5c/(5c) = 0.9  ->  Q90（下调区）
        F_up   = 1 - 1.5c/(5c) = 0.7  ->  Q70（上调区）
    对已锁定计划的逐段最优即为 Q70、Q90 与 q0（拐点）的中位数。
    """
    net = fl_d[h0:] - fc_dk[h0:]
    if scen is None:
        return np.maximum(net, q0)
    q_hi = np.quantile(scen, q_down, axis=0)
    q_lo = np.quantile(scen, q_up, axis=0)
    target = np.median(np.stack([q_lo, q_hi, np.asarray(q0, float)]), axis=0)
    return np.maximum(net, target)
```

**4. 调整购电 LP**

```python
def solve_adjustment(q0, risk_net, prices, initial, terminal_value):
    """重解调整购电 a；与计划 q0 的偏差经 p、q 变量按分段净结算计入目标。"""
    n = len(risk_net)
    q0 = np.asarray(q0, float)

    def idx(block, t):
        return block * n + t

    # 变量 a, C, D, U, E, p, q（7n）：平衡与 SOC 方程同日前 LP
    a_eq = lil_matrix((2 * n, 7 * n))
    b_eq = np.zeros(2 * n)
    for t in range(n):
        a_eq[t, idx(0, t)] = 1
        a_eq[t, idx(1, t)] = -1
        a_eq[t, idx(2, t)] = 1
        a_eq[t, idx(3, t)] = -1
        b_eq[t] = risk_net[t]
        a_eq[n + t, idx(4, t)] = 1
        a_eq[n + t, idx(1, t)] = -ETA
        a_eq[n + t, idx(2, t)] = 1.0 / ETA
        if t:
            a_eq[n + t, idx(4, t - 1)] = -1
        b_eq[n + t] = initial if t == 0 else 0.0

    # 偏差线性化：p >= q0 - a，q >= a - q0
    a_ub = lil_matrix((2 * n, 7 * n))
    b_ub = np.zeros(2 * n)
    for t in range(n):
        a_ub[2 * t, idx(5, t)] = -1
        a_ub[2 * t, idx(0, t)] = -1
        b_ub[2 * t] = -q0[t]
        a_ub[2 * t + 1, idx(6, t)] = -1
        a_ub[2 * t + 1, idx(0, t)] = 1
        b_ub[2 * t + 1] = q0[t]

    bounds = (
        [(0, None)] * n + [(0, S)] * n + [(0, S)] * n + [(0, None)] * n
        + [(EMIN, EMAX)] * n + [(0, None)] * n + [(0, None)] * n
    )

    # 目标：c*(a + 0.5*(p + q)) - nu*E_end，即分段净结算 + 期末电量价值
    obj = np.zeros(7 * n)
    obj[:n] = prices
    obj[idx(5, 0): idx(5, 0) + n] = 0.5 * prices
    obj[idx(6, 0): idx(6, 0) + n] = 0.5 * prices
    obj[idx(4, n - 1)] = -terminal_value

    first = linprog(obj, A_eq=a_eq.tocsr(), b_eq=b_eq, A_ub=a_ub.tocsr(),
                    b_ub=b_ub, bounds=bounds, method="highs")
    if not first.success:
        raise RuntimeError(first.message)
    # 第二阶段：成本容差内消除无效充放吞吐
    throughput = np.zeros(7 * n)
    throughput[idx(1, 0): idx(1, 0) + n] = 1
    throughput[idx(2, 0): idx(2, 0) + n] = 1
    second = linprog(throughput, A_eq=a_eq.tocsr(), b_eq=b_eq,
                     A_ub=csr_matrix(np.vstack([a_ub.toarray(), obj])),
                     b_ub=np.r_[b_ub, first.fun + 1e-6],
                     bounds=bounds, method="highs")
    if not second.success:
        raise RuntimeError(second.message)
    x = second.x.copy()
    x[np.abs(x) < 1e-8] = 0
    return x[:n], x[idx(4, 0): idx(4, 0) + n], float(first.fun)
```

**5. 分段储备规则的编译执行内核**

```python
@njit(cache=True)
def _run_rule_kernel(net, q, ref, prices, s0s, s1s, has_lam, deltas,
                     lams, signals, initial, emin, emax, s, eta):
    """编译后的分段截断仿射储备仿真，与 numpy 参考实现逐位一致。"""
    m = net.shape[0]
    n_stages = s0s.shape[0]
    energy = np.full(m, initial)
    emergency_cost = np.zeros(m)
    for j in range(n_stages):
        for t in range(s0s[j], s1s[j]):
            for i in range(m):
                reserve = ref[t] + deltas[j] + lams[j] * signals[i, j]
                if reserve < emin:
                    reserve = emin
                elif reserve > emax:
                    reserve = emax
                r = net[i, t] - q[t]
                charge = min(min(max(-r, 0.0), s),
                             max((emax - energy[i]) / eta, 0.0))
                discharge = min(min(max(r, 0.0), s),
                                eta * max(energy[i] - reserve, 0.0))
                emergency_cost[i] += 5.0 * prices[i, t] * max(r - discharge, 0.0)
                energy[i] += eta * charge - discharge / eta

    """w_i = exp(-(d-1-i)/tau) 归一化权重。"""
    idx = np.arange(first, d, dtype=float)
    w = np.exp(-(d - 1 - idx) / tau)

def price_weighted_quantile(net, price, w, alpha):
    """价格加权 alpha 分位数：Q^p = inf{x : sum(w*p*1(N<=x)) / sum(w*p) >= alpha}。"""
    m, n = net.shape
    out = np.zeros(n, float)
    for j in range(n):
        order = np.argsort(net[:, j], kind="stable")
        ns, ps, ws = net[order, j], price[order, j], w[order]
        total = float(ws @ ps)
        if total <= 0.0:
            out[j] = float(np.quantile(net[:, j], alpha))
            continue
        cum = np.cumsum(ws * ps)
        k = int(np.searchsorted(cum, alpha * total, side="left"))
        out[j] = ns[min(k, m - 1)]
    return out


def scenario_mean_price(price_scen, w):
    """情景均值决策价 pbar_t = sum(w*p_t) / sum(w)。"""
    return (w[:, None] * price_scen).sum(axis=0) / float(w.sum())
```
